104 inline TimeUnitType makeTimeUnit(TimeUnit unit, qreal customSec = 1.0) {
105 constexpr qreal SECONDS_IN_MINUTE = 60.0;
106 constexpr qreal SECONDS_IN_HOUR = 60.0 * SECONDS_IN_MINUTE; // 3600
107 constexpr qreal SECONDS_IN_DAY = 24.0 * SECONDS_IN_HOUR; // 86400
108 constexpr qreal SECONDS_IN_WEEK = 7.0 * SECONDS_IN_DAY; // 604800
109 constexpr qreal SECONDS_IN_MONTH = 30.4375 * SECONDS_IN_DAY; // ≈2628000
110 constexpr qreal SECONDS_IN_YEAR = 365.25 * SECONDS_IN_DAY; // ≈31536000
111 switch (unit) {
112 case TimeUnit::SECOND:
113 return {unit, 1.0};
114 case TimeUnit::MINUTE:
115 return {unit, SECONDS_IN_MINUTE};
116 case TimeUnit::HOUR:
117 return {unit, SECONDS_IN_HOUR};
118 case TimeUnit::DAY:
119 return {unit, SECONDS_IN_DAY};
120 case TimeUnit::WEEK:
121 return {unit, SECONDS_IN_WEEK};
122 case TimeUnit::MONTH:
123 return {unit, SECONDS_IN_MONTH};
124 case TimeUnit::YEAR:
125 return {unit, SECONDS_IN_YEAR};
126 case TimeUnit::CUSTOM:
127 return {unit, customSec};
128 default:
129 return {unit, 1.0};
130 }
131 }

133 // Time parser, can parse 1d, 1h, 1m, 1s, 1w, 1m, 1y to convert it into seconds
134 inline qreal parseTime(const QString& str, bool* okay = nullptr) {
135 if (okay != nullptr) {
136 *okay = false;
137 }
138 qreal result = 0.0;
139 QString cleanStr = str.simplified();
140 if (cleanStr.isEmpty()) {
141 return result;
142 }
143
144 // Try to parse the date (e.g., "2025-04-27")
145 QDateTime dti = QDateTime::fromString(cleanStr, "yyyy-MM-dd");
146 if (dti.isValid()) {
147 if (okay != nullptr) {
148 *okay = true;
149 }
150 return (qreal)dti.toSecsSinceEpoch();
151 }
152
153 // Parsing time string ("1d 2h")
154 // qDebug() << "Parsing time string: " << cleanStr;
155 QStringList segments = cleanStr.split(QRegExp("\\s+"), QString::SkipEmptyParts);
156 // qDebug() << "Segments: " << segments;
157 bool parsedAny = false;
158 QRegExp rxe(R"((-?\d+(?:\.\d+)?)([a-zA-Z]+)?)");
159 for (const QString& segment : segments) {
160 if (rxe.indexIn(segment) != -1) {
161 bool convertOk;
162 qreal value = rxe.cap(1).toDouble(&convertOk);
163 if (!convertOk) {
164 continue;
165 }
166 QString unit = rxe.cap(2).toLower();
167 // qDebug() << "Value: " << value << " Unit: " << unit;
168 TimeUnitType type = makeTimeUnit(TimeUnit::NONE);
169 if (unit.isEmpty()) {
170 type = makeTimeUnit(TimeUnit::NONE);
171 } else if (unit == "s") {
172 type = makeTimeUnit(TimeUnit::SECOND);
173 } else if (unit == "m") {
174 type = makeTimeUnit(TimeUnit::MINUTE);
175 } else if (unit == "h") {
176 type = makeTimeUnit(TimeUnit::HOUR);
177 } else if (unit == "d") {
178 type = makeTimeUnit(TimeUnit::DAY);
179 } else if (unit == "w") {
180 type = makeTimeUnit(TimeUnit::WEEK);
181 } else if (unit == "mo") {
182 type = makeTimeUnit(TimeUnit::MONTH);
183 } else if (unit == "y") {
184 type = makeTimeUnit(TimeUnit::YEAR);
185 } else {
186 continue;
187 }
188 result += type.toSeconds(value);
189 parsedAny = true;
190 }
191 }
192
193 if (okay != nullptr && parsedAny) {
194 *okay = true;
195 }
196 return result;
197 }
